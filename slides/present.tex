\section{Cavity--matter interaction}
%=============================================================================
\begin{frame}{Molecules in an optical cavity: polaritons}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \small
      \begin{itemize}
        \item A Fabry--P\'erot cavity confines the electromagnetic field into
              discrete \textbf{standing-wave modes} of frequency $\omega_c$ set
              by the mirror spacing $L$.
        \item When the light--matter coupling exceeds all loss rates
              (\emph{strong coupling}), molecule and photon form hybrid
              eigenstates: \textbf{polaritons}.
        \item These split into a \textbf{lower} (LP) and \textbf{upper} (UP)
              polariton, separated by the \textbf{Rabi splitting} $\Omega$.
        \item Coupling $N$ molecules collectively enhances the interaction as
              $\sqrt{N}$: the single-molecule coupling is
              $g=\Omega/(2\sqrt N)$.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95]
        % mirrors
        \draw[very thick,gray] (0,0) -- (0,3);

        \node[capgray] at (3.2,-0.3) {\tiny mirror};

\section{Jahn--Teller active molecules}
%=============================================================================
\begin{frame}{The Jahn--Teller effect and the $(E\times e)$ model}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \small
      \textbf{Jahn--Teller theorem.} A non-linear molecule in an orbitally
      degenerate electronic state spontaneously distorts to lift the
      degeneracy.\\[0.4em]
      \textbf{$(E\times e)$ problem.} A doubly-degenerate electronic state $E$
      (components $|E_x\rangle,|E_y\rangle$) coupled to a doubly-degenerate
      vibrational mode $e$ (coordinates $Q_x,Q_y$):
      \begin{itemize}
        \item ground state $|A\rangle$, excited doublet $|E_\pm\rangle=(|E_x\rangle\pm i|E_y\rangle)/\sqrt2$;
        \item potential surfaces form a \emph{Mexican hat} with a
              \textbf{conical intersection};
        \item vibronic coupling $\kappa$ mixes electronic and nuclear motion
              $\Rightarrow$ \textbf{vibronic angular momentum} is exchanged.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.9]
        % mexican hat cross-section
        \draw[->] (-2.2,0) -- (2.4,0) node[right]{\scriptsize $Q$};
        \draw[->] (0,-0.3) -- (0,2.8) node[above]{\scriptsize $E$};
        \draw[thick,red!70,domain=-2:2,samples=100,smooth]
          plot (\x,{0.9*(\x*\x*0.55 - 0.9) + 1.7 + 0.15*abs(\x)});
        \draw[thick,red!70,domain=-2:2,samples=100,smooth]
          plot (\x,{0.9*(\x*\x*0.55 - 0.9) + 1.7 - 0.15*abs(\x)});
        \node[red!70] at (1.7,2.5) {\scriptsize $|E_\pm\rangle$};
        % ground
        \draw[thick,blue!70,domain=-1.6:1.6,samples=80,smooth]
          plot (\x,{0.35*\x*\x});
        \node[blue!70] at (1.5,0.7) {\scriptsize $|A\rangle$};
        \draw[<->,gray] (0.05,0.9) -- (0.05,1.55);
        \node[capgray] at (0.5,1.25) {\tiny $\omega_c$};
      \end{tikzpicture}\\[0.3em]
      \figcap{Excited $(E\times e)$ surfaces (red) and ground $|A\rangle$
      (blue) vs.\ mode coordinate $Q$.}
    \end{column}
  \end{columns}
\end{frame}

\section{Single molecule: cavity Jahn--Teller polaritons (2023)}
%=============================================================================
\begin{frame}{Cavity Jahn--Teller Hamiltonian --- one molecule}
  \small
  The molecule couples to two perpendicular normal-incidence cavity modes,
  recast as circular modes $\hat a_\pm$ carrying photon angular momentum
  $\pm\hbar$. The cavity--JT Hamiltonian is
  $\hat H=\hat H_{\mathrm{JT}}+\hat H_{C}+\hat H_{CM}$, with
  \[
    \hat H_{\mathrm{JT}}=\!\!\sum_{r=+,-}\!\!\big[\hbar\omega\,\hat b_r^\dagger \hat b_r
      + \epsilon\,|E_r\rangle\langle E_r|\big]
      + \kappa\big[(\hat b_+^\dagger+\hat b_-)\,|E_-\rangle\langle E_+| + \text{h.c.}\big],
  \]
  \[
    \hat H_{C}=\hbar\omega_c\big(\hat a_+^\dagger \hat a_+ + \hat a_-^\dagger \hat a_-\big),
    \qquad
    \hat H_{CM}=\frac{\Omega}{2}\big(\hat a_+^\dagger |A\rangle\langle E_+|
      + \hat a_-^\dagger |A\rangle\langle E_-| + \text{h.c.}\big).
  \]
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \scriptsize
      \textbf{Operators.}
      $\hat b_\pm=(\hat b_x\mp i\hat b_y)/\sqrt2$ --- ladder operators of the
      degenerate $e$ vibration carrying $\pm\hbar$ vibrational angular momentum;
      $\hat a_\pm$ --- annihilation operators of the two circularly-polarized
      ($\pm\hbar$) cavity photons.\\[0.4em]
      The JT (vibronic) coupling $\kappa$ exchanges angular momentum
      \emph{between the electronic and vibrational subsystems of the molecule};
      the cavity coupling $\hat H_{CM}$ then exchanges it with the photon.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Angular momenta \& conservation}
      \scriptsize
      \vspace{-0.7em}
      \begin{align*}
        \hat L_z &= \hbar\big(\hat b_+^\dagger\hat b_+-\hat b_-^\dagger\hat b_-\big)
          &&\text{vibrational}\\
        \hat S_z &= \hbar\big(|E_+\rangle\langle E_+|-|E_-\rangle\langle E_-|\big)
          &&\text{electronic}\\
        \hat l_z &= \hbar\big(\hat a_+^\dagger\hat a_+-\hat a_-^\dagger\hat a_-\big)
          &&\text{photonic}
      \end{align*}
      \vspace{-0.6em}
      \[ \boxed{\ \hat J = 2\hat L_z + \hat S_z + \hat l_z\ }\ \text{is conserved.} \]
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

      \centering
      \includegraphics[width=\linewidth]{fig_1mol_sticks.png}

\section{Collective vibronic cascade (2025)}
%=============================================================================
\begin{frame}{$N$ molecules in a common cavity}
  \small
  $N$ identical JT molecules coupled to the two cavity modes
  ($\hat H_{m,k}$ is the single-molecule JT Hamiltonian of the previous slide):
  \[
    \hat H=\hat H_c+\sum_{k=1}^{N}\big(\hat H_{m,k}+\hat H_{m\text{-}c,k}\big),\qquad
    \hat H_c=\hbar\omega_c\big(\hat a_+^\dagger\hat a_++\hat a_-^\dagger\hat a_-\big),
  \]
  \[
    \hat H_{m\text{-}c,k}=\frac{\Omega}{2\sqrt N}
      \big(\hat a_+^\dagger|A\rangle\langle E_+|_k+\hat a_-^\dagger|A\rangle\langle E_-|_k+\text{h.c.}\big).
  \]
  The $\sqrt N$ normalization keeps the collective coupling
  $g=\Omega/(2\sqrt N)$ physical. Two conserved quantities:
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Excitation number}
      \vspace{-0.6em}
      \[\hat N_{\mathrm{ex}}=\sum_k\!\big(|E_+\rangle\langle E_+|_k+|E_-\rangle\langle E_-|_k\big)+\hat a_+^\dagger\hat a_++\hat a_-^\dagger\hat a_-\]
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Total angular momentum}
      \vspace{-0.6em}
      \[\hat J=\sum_{k=1}^{N}\hat V_k+\hat P,\qquad \hat V_k=2\hat L_{z,k}+\hat S_{z,k}\]
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  We work in the optically-relevant single-excitation sector
  $(n_{\mathrm{ex}}=1,\ j=-1)$.
\end{frame}

\begin{frame}{The collective vibronic cascade}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      Each molecule's JT Hamiltonian has vibronic eigenstates with a definite
      \textbf{vibronic angular momentum} $v=2(n_+-n_-)+S_z$:
      \[\hat H_{m,k}=\sum_v\sum_i \lambda_i^{v}\,|\lambda_i^{v}\rangle\langle\lambda_i^{v}|_k\]
      \begin{itemize}
        \item \textbf{$N=1$:} bounded --- starting from the ground state only
              $v\in\{0,-1,-2\}$ are reachable.
        \item \textbf{$N\ge 2$:} the cavity photon shuttles angular momentum
              \emph{between} molecules $\Rightarrow$ an unbounded ladder of $v$
              sectors --- the \textbf{collective vibronic cascade}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.72,every node/.style={font=\scriptsize}]
        \node[draw,rounded corners,fill=orange!15] (a) at (0,0) {$|0,-1\rangle$};
        \node[draw,rounded corners,fill=orange!15] (b) at (0,-1) {$|-1,0\rangle$};
        \node[draw,rounded corners,fill=orange!15] (c) at (0,-2) {$|-2,1\rangle$};
        \draw[<->] (a)--(b); \draw[<->] (b)--(c);
        \node at (0,-2.7) {$N=1$ (bounded)};
        \node[draw,rounded corners,fill=green!12] (x) at (3,0) {$|0,0,-1\rangle$};
        \node[draw,rounded corners,fill=green!12] (y) at (3,-1) {$|-1,0,0\rangle$};
        \node[draw,rounded corners,fill=green!12] (z) at (3,-2) {$|-2,1,0\rangle$};
        \node[draw,rounded corners,fill=green!12] (w) at (3,-3) {$|-6,5,0\rangle\ \cdots$};
        \draw[<->] (x)--(y); \draw[<->] (y)--(z); \draw[<->] (z)--(w);
        \node at (3,-3.7) {$N=2$ (cascade)};
      \end{tikzpicture}\\[0.4em]
      \figcap{The basis-state chains above show the sectors populated
      starting from the ground state $+$ one RCP photon.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Participation ratio and the $P(v)$ heatmap}
  \small
  To \emph{quantify} how far the cascade spreads, we decompose each polaritonic
  eigenstate $|\psi\rangle=\sum_{a,b,p}C_{a,b,p}\,|a,b,p\rangle$ over
  single-molecule vibronic sectors $v$:
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Sector population $P(v)$}
      \vspace{-0.6em}
      \[ P^{(k)}(v_0)=\!\!\sum_{a:\,v_a=v_0}\ \sum_{b,p}|C_{a,b,p}|^2,
         \qquad \sum_v P(v)=1. \]
      \end{block}
      \begin{block}{Participation ratio}
      \vspace{-0.6em}
      \[ \mathrm{PR}=\frac{1}{\sum_v P(v)^2} \]
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \scriptsize
      \begin{itemize}
        \item $\mathrm{PR}$ counts how many $v$ sectors meaningfully
              participate: $\mathrm{PR}=1$ if localized in one sector, larger
              if spread over many.
        \item \textbf{$N=1$:} $\mathrm{PR}\le 3$ (bounded).\quad
              \textbf{$N=2$:} $\mathrm{PR}$ up to $\sim 20$.
        \item The \textbf{$P(v)$ heatmap} plots, for every bright polaritonic
              state (x, energy-ordered), its $P(v)$ over sectors $v$ (y),
              colored by value --- the paper's Fig.~S1.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Introducing disorder}
%=============================================================================
\begin{frame}{Why disorder? Two guiding papers}
  \small
  Real molecular ensembles are \textbf{inhomogeneous}: each molecule has a
  slightly different transition energy (static \emph{energy disorder}). Two
  works frame what disorder does to a cavity-coupled ensemble:
  \begin{block}{(a) Arrowhead-matrix picture (single excitation)}
  \scriptsize
  In the single-excitation sector the disordered Tavis--Cummings Hamiltonian is
  an \textbf{arrowhead matrix}: a disordered diagonal of molecular energies plus
  one coupling row/column to the photon. Its eigenstates are two
  \textbf{polariton outliers} (split by $\pm\sqrt N g$) plus $N-1$
  \textbf{dark states}; disorder induces \emph{semi-localization} and reshapes
  the photon spectral function.
  \end{block}
  \begin{block}{(b) Disordered Holstein--Tavis--Cummings (with vibrations)}
  \scriptsize
  Adds vibrations (Holstein coupling). Central result: \textbf{disorder
  enhances} excitation transfer and vibrational entanglement, and dark states
  \emph{acquire photon weight} (brighten). Effect peaks when disorder
  $W\sim g_c$.
  \end{block}
\end{frame}

\begin{frame}{Disorder: the equations}
  \small
  \textbf{Disordered Holstein--Tavis--Cummings model} (Wellnitz \textit{et al.}):
  \[
    \hat H=\hat H_{\mathrm{TC}}+\hat H_{\mathrm{vib}}+\hat H_{\mathrm{H}}+\hat H_{\mathrm{dis}},
    \qquad
    \boxed{\ \hat H_{\mathrm{dis}}=\sum_n \epsilon_n\,\hat\sigma_n^+\hat\sigma_n^-\ }
  \]
  with i.i.d.\ energy offsets $\epsilon_n$ of mean $0$ and variance $W^2$
  (Gaussian). Consequences quantified in the papers:
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}\scriptsize
        \item Dark states gain photon weight
              $\sum_{\mathrm{dark}}|\langle d|1_{\mathrm{ph}}\rangle|^2\approx W^2/2g_c^2$
              $\Rightarrow$ they \textbf{appear in absorption}.
        \item Polaritons stay protected while $g$ beats disorder: crossover at
              $W\sim 2g$.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}\scriptsize
        \item Arrowhead: photon spectral function
              $A(\varepsilon)=\sum_a \mathrm{PW}_a\,\delta(\varepsilon-\varepsilon_a)$
              --- this \emph{is} the absorption spectrum.
        \item Eigenstate amplitudes $\psi_{a,j}\sim 1/(\varepsilon_a-\omega_j)^2$;
              level statistics Poisson $\to$ semi-Poisson.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Disorder in our model \& results}
%=============================================================================
\begin{frame}{How disorder enters our model}
  \small
  We add \textbf{Gaussian energy disorder} to the two JT molecules'
  transition energies:
  \[
    \epsilon_i \sim \mathcal N(\epsilon,\ \sigma^2),\qquad i=1,2,\qquad \epsilon=7.0~\text{eV}.
  \]
  Per disorder \emph{realization} we build $\hat H$ in the
  $(n_{\mathrm{ex}}=1,\ j=-1)$ sector, diagonalize, take the intensity of each
  eigenstate as $|\langle i_0|\psi\rangle|^2$, broaden with a Lorentzian (FWHM $\gamma=0.025$~eV), and
  \textbf{average over realizations}.
  \begin{block}{Energy scales}
  \centering\scriptsize
  \begin{tabular}{lll}
    \toprule
    vibrational quantum $\omega$ & $0.082$~eV & JT coupling $\kappa/\omega=2.2$\\
    per-molecule coupling $g=\Omega/2\sqrt N$ & $0.242$~eV & cavity $\omega_c=6.85$~eV\\
    Rabi splitting $\Omega=2\sqrt N g$ & $0.685$~eV & molecular excitation energies $\epsilon=7.0$~eV, $N=2$\\
    \bottomrule
  \end{tabular}
  \end{block}
  Disorder strength quoted relative to $g$: we run at $\sigma\approx 2g$
  (strong-disorder band), where the representative cases are physically distinct.
\end{frame}

\begin{frame}{Disorder realizations: the Gaussian distribution}
  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      \centering
      \includegraphics[width=\linewidth]{disorder_gaussian_sigma0.48.png}
    \end{column}
    \begin{column}{0.38\textwidth}
      \small
      \begin{itemize}
        \item Each molecule's excitation energy is drawn from
              $\mathcal N(\epsilon,\sigma^2)$ with $\sigma=0.48$~eV.
        \item $\sigma/g\approx 2.0$, $\sigma/\Omega\approx 0.70$ --- just below
              the $W=2g$ crossover.
        \item Dashed lines / arrow mark $\pm\sigma$ ($-0.48$ to $+0.48$~eV).

\caseframe{A}{resonant baseline (no disorder)}{%
  Starting point: both molecules essentially on resonance,
  $\epsilon_1\!\approx\!\epsilon_2\!\approx\!\epsilon$.}{%
  $(\epsilon_1,\epsilon_2)=(6.98,7.01)$, $\Delta\!=\!0.02$, $s\!\approx\!0$;
  \textbf{39 bright}, $\mathrm{PR}_{\max}=20.6$ at $E=7.361$~eV --- reproduces
  the clean collective cascade.}

\caseframe{D}{common blue detuning}{%
  From A, shift \emph{both} molecules \emph{up} together by $\sim\!+g$ (matched,
  $\Delta$ small): a common blue shift.}{%
  $(\epsilon_1,\epsilon_2)=(7.33,7.17)$, $\Delta\!=\!0.17$, $s\!=\!+1.0g$;
  \textbf{12 bright}, $\mathrm{PR}_{\max}=12.5$ at $E=7.354$~eV --- \textbf{partial}:
  bright manifold pushed partly off the photon.}

\caseframe{E}{common red detuning (matched)}{%
  Flip the common shift to $-g$ (mirror of D, but red): this lands in the
  enhancement band, pushing the bright polariton \emph{into} the dense dark
  manifold.}{%
  $(\epsilon_1,\epsilon_2)=(6.67,6.70)$, $\Delta\!=\!0.03$, $s\!=\!-1.3g$;
  \textbf{104 bright}, $\mathrm{PR}_{\max}=22.4$ at $E=7.215$~eV --- \textbf{enhanced}:
  disorder brightens the whole dark band.}

\caseframe{F}{single-molecule resonant}{%
  Now introduce a real inter-molecular mismatch: bring one molecule back
  \emph{onto} resonance while the other stays detuned by $\gtrsim\!g$ --- a
  mildly asymmetric \emph{collective} case (not localized).}{%
  $(\epsilon_1,\epsilon_2)=(7.01,6.75)$, $\Delta\!=\!g$, $s\!=\!-0.5g$;
  \textbf{132 bright}, $\mathrm{PR}_{\max}=21.6$ at $E=7.424$~eV --- \textbf{enhanced}
  (maximal dark-state brightening).}

\caseframe{H}{A--C line, interior (opposite disorder)}{%
  Detune the two molecules \emph{oppositely} (one red, one blue) about
  resonance, $s\!\approx\!0$ --- the interior of the line joining A and C.}{%
  $(\epsilon_1,\epsilon_2)=(6.72,7.33)$, $\Delta\!=\!2.5g$, $s\!\approx\!0$;
  \textbf{46 bright}, $\mathrm{PR}_{\max}=16.2$ at $E=7.337$~eV --- \textbf{intermediate}:
  brightens more peaks than A yet spreads each less.}

\caseframe{C}{opposite disorder (far)}{%
  Push the opposite detuning \emph{further out} along the same A--C line
  ($\Delta\!=\!6g$): both molecules now leave the bright line.}{%
  $(\epsilon_1,\epsilon_2)=(6.28,7.76)$, $\Delta\!=\!6g$, $s\!\approx\!0$;
  \textbf{14 bright}, $\mathrm{PR}_{\max}=8.5$ at $E=7.728$~eV --- \textbf{partial$\to$dead}:
  cascade collapses to a near-bare cavity peak.}

\caseframe{B}{large same-side mismatch (localized)}{%
  Instead of opposite, detune one molecule far on the \emph{same} side
  ($\Delta\!=\!4g$): the excitation \emph{localizes} on the resonant molecule
  $\Rightarrow$ spectrum $\approx$ a single JT-polariton (orange: 1-mol
  $g/\sqrt2$).}{%
  $(\epsilon_1,\epsilon_2)=(8.08,7.08)$, $\Delta\!=\!4g$, $s\!=\!+2.4g$;
  \textbf{11 bright}, $\mathrm{PR}_{\max}=3.8$ at $E=7.921$~eV --- \textbf{dead}
  (single-molecule bound).}

\caseframe{G}{bare-molecule limit (decoupled)}{%
  Finally push \emph{both} molecules far below resonance together
  ($|s|\!\gg\!\sqrt N g$): both decouple from the cavity.}{%
  $(\epsilon_1,\epsilon_2)=(6.25,6.15)$, $\Delta\!=\!0.10$, $s\!=\!-3.3g$;
  \textbf{8 bright}, $\mathrm{PR}_{\max}=4.5$ at $E=6.118$~eV --- \textbf{dead}
  (near-bare photon).}

\begin{frame}{Summary}
  \small
  \begin{enumerate}
    \item \textbf{Reproduced} the single-molecule cavity-JT polariton spectrum
          (LP/UP, bounded $v$) and the two-molecule collective spectrum +
          $P(v)$ heatmap (unbounded cascade, $\mathrm{PR}\sim20$).
    \item \textbf{Introduced disorder} following the arrowhead-matrix and
          disordered Holstein--Tavis--Cummings frameworks: Gaussian excitation-energy
          disorder $\epsilon_i\sim\mathcal N(\epsilon,\sigma^2)$, dark states
          brighten, cascade protected while $g$ beats $\sigma$.
    \item \textbf{Implemented \& analyzed} disorder in our $N=2$ model:
          realizations, representative cases (A--H), and $P(v)$ heatmaps with an
          LP/UP energy overlay.
    \item \textbf{Key result:} the number of bright vibronic states and the
          cascade PR track the two disorder axes --- \emph{enhanced} (E, F, H),
          \emph{partial}, or \emph{dead} (B, G).
  \end{enumerate}
\end{frame}
