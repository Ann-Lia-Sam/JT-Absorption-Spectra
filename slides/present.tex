\section{Cavity--matter interaction}
%=============================================================================
\begin{frame}{Molecules in an optical cavity: polaritons}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \small
      \begin{itemize}
        \item A Fabry-P\'erot cavity confines the electromagnetic field into
              discrete \textbf{standing-wave modes} of frequency $\omega_c$ set
              by the mirror spacing $L$.
        \item When the light-matter coupling exceeds all loss rates
              (\emph{strong coupling}), molecule and photon form hybrid
              eigenstates: \textbf{polaritons}.\refmark{6}
        \item These split into a \textbf{lower} (LP) and \textbf{upper} (UP)
              polariton, separated by the \textbf{Rabi splitting} $\Omega$.
        \item Coupling $N$ molecules collectively enhances the interaction as
              $\sqrt{N}$: the single-molecule coupling is
              $g=\Omega/(2\sqrt N)$.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering

\section{Jahn--Teller active molecules}
%=============================================================================
\begin{frame}{The Jahn--Teller effect and the $(E\times e)$ model}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \small
      \textbf{Jahn-Teller theorem.}\refmark{5} A non-linear molecule in an orbitally
      degenerate electronic state spontaneously distorts to lift the
      degeneracy.\\[0.4em]
      \textbf{$(E\times e)$ problem.} A doubly-degenerate electronic state $E$
      (components $|E_x\rangle,|E_y\rangle$) coupled to a doubly-degenerate
      vibrational mode $e$ (coordinates $Q_x,Q_y$):
      \begin{itemize}
        \item ground state $|A\rangle$, excited doublet $|E_\pm\rangle=(|E_x\rangle\pm i|E_y\rangle)/\sqrt2$;
        \item potential surfaces form a \emph{Mexican hat} with a
              \textbf{conical intersection};
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering

\end{frame}

\section{Single molecule: cavity Jahn--Teller polaritons (2023)}
%=============================================================================
\begin{frame}{Cavity Jahn--Teller Hamiltonian - One Molecule}
  \small
 There are circular cavity modes $\hat a_\pm$ carrying photon angular momentum
  $\pm\hbar$.\refmark{1} The cavity--JT Hamiltonian is
  $\hat H=\hat H_{\mathrm{JT}}+\hat H_{C}+\hat H_{CM}$, with
  \[
    \hat H_{\mathrm{JT}}=\!\!\sum_{r=+,-}\!\!\big[\hbar\omega\,\hat b_r^\dagger \hat b_r
      + \epsilon\,|E_r\rangle\langle E_r|\big]
      + \kappa\big[(\hat b_+^\dagger+\hat b_-)\,|E_-\rangle\langle E_+| + \text{h.c.}\big],
  \]
  \[
    \hat H_{C}=\hbar\omega_c\big(\hat a_+^\dagger \hat a_+ + \hat a_-^\dagger \hat a_-\big),
    \qquad
    \hat H_{CM}=\frac{\Omega}{2}\big(\hat a_+^\dagger |A\rangle\langle E_+|
      + \hat a_-^\dagger |A\rangle\langle E_-| + \text{h.c.}\big).
  \]
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \scriptsize
      \textbf{Operators.}
      $\hat b_\pm=(\hat b_x\mp i\hat b_y)/\sqrt2$ - ladder operators of the
      degenerate $e$ vibration carrying $\pm\hbar$ vibrational angular momentum;
      $\hat a_\pm$ - annihilation and creation operators of the two circularly-polarized
      ($\pm\hbar$) cavity photons.\\[0.4em]
      The JT (vibronic) coupling $\kappa$ exchanges angular momentum
      \emph{between the electronic and vibrational subsystems of the molecule};
      the cavity coupling $\hat H_{CM}$ then exchanges it with the photon.
      Vibronic coupling $\kappa$ mixes electronic and nuclear motion
      $\Rightarrow$ vibronic angular momentum is exchanged.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Angular momenta \& conservation}
      \scriptsize
      \vspace{-0.7em}
      \begin{align*}
        \hat L_z &= \hbar\big(\hat b_+^\dagger\hat b_+-\hat b_-^\dagger\hat b_-\big)
          &&\text{vibrational}\\
        \hat S_z &= \hbar\big(|E_+\rangle\langle E_+|-|E_-\rangle\langle E_-|\big)
          &&\text{electronic}\\
        \hat l_z &= \hbar\big(\hat a_+^\dagger\hat a_+-\hat a_-^\dagger\hat a_-\big)
          &&\text{photonic}
      \end{align*}
      \vspace{-0.6em}
      \[ \boxed{\ \hat J = 2\hat L_z + \hat S_z + \hat l_z\ }\ \text{is conserved.} \]
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Collective vibronic cascade (2025)}
%=============================================================================
\begin{frame}{Collective Cavity Coupling($N$ molecules)}
  \small
  $N$ identical JT molecules\refmark{2} coupled to the two cavity modes
  ($\hat H_{m,k}$ is the single-molecule JT Hamiltonian of the previous slide):
  \[
    \hat H=\hat H_c+\sum_{k=1}^{N}\big(\hat H_{m,k}+\hat H_{m\text{-}c,k}\big),\qquad
    \hat H_c=\hbar\omega_c\big(\hat a_+^\dagger\hat a_++\hat a_-^\dagger\hat a_-\big),
  \]
  \[
    \hat H_{m\text{-}c,k}=\frac{\Omega}{2\sqrt N}
      \big(\hat a_+^\dagger|A\rangle\langle E_+|_k+\hat a_-^\dagger|A\rangle\langle E_-|_k+\text{h.c.}\big).
  \]
   Two conserved quantities:
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Excitation number}
      \vspace{-0.6em}
      \[\hat N_{\mathrm{ex}}=\sum_k\!\big(|E_+\rangle\langle E_+|_k+|E_-\rangle\langle E_-|_k\big)+\hat a_+^\dagger\hat a_++\hat a_-^\dagger\hat a_-\]
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Total angular momentum}
      \vspace{-0.6em}
      \[\hat J=\sum_{k=1}^{N}\hat V_k+\hat P,\qquad \hat V_k=2\hat L_{z,k}+\hat S_{z,k}\]
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  We work in the optically-relevant single-excitation sector
  $(n_{\mathrm{ex}}=1,\ j=-1)$.
\end{frame}

\begin{frame}{The Collective Vibronic Cascade}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      Each molecule's JT Hamiltonian has vibronic eigenstates with a definite
      \textbf{vibronic angular momentum} $v=2(n_+-n_-)+S_z$, where $n_+,n_-$
      are the vibrational quanta (Fock occupation numbers) of the $b_+,b_-$
      modes:
      \[\hat H_{m,k}=\sum_v\sum_i \lambda_i^{v}\,|\lambda_i^{v}\rangle\langle\lambda_i^{v}|_k\]
      \begin{itemize}
        \item \textbf{$N=1$:} bounded - starting from the ground state only
              $v\in\{0,-1,-2\}$ are reachable.
        \item \textbf{$N\ge 2$:} the cavity photon mediates the transfer of angular momentum between molecules
             $\Rightarrow$ an unbounded ladder of $v$
              sectors - the \textbf{collective vibronic cascade}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.72,every node/.style={font=\scriptsize}]
        \node[draw,rounded corners,fill=orange!15] (a) at (0,0) {$|0,-1\rangle$};

\end{frame}

\begin{frame}{Participation Ratio and the $P(v)$ Heatmap}
  \small
  To \emph{quantify} how far the cascade spreads, we decompose each polaritonic
  eigenstate $|\psi\rangle=\sum_{a,b,p}C_{a,b,p}\,|a,b,p\rangle$ over
  single-molecule vibronic sectors $v$:
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Sector Population $P(v)$}
      \vspace{-0.6em}
      \[ P^{(k)}(v_0)=\!\!\sum_{a:\,v_a=v_0}\ \sum_{b,p}|C_{a,b,p}|^2,
         \qquad \sum_v P(v)=1. \]
      \end{block}
      \begin{block}{Participation Ratio}
      \vspace{-0.6em}
      \[ \mathrm{PR}=\frac{1}{\sum_v P(v)^2} \]
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \scriptsize
      \begin{itemize}
        \item $\mathrm{PR}$ counts how many $v$ sectors of an eigenstate participate meaningfully: $\mathrm{PR}=1$ if localized in one sector, larger
              if spread over many.
        \item \textbf{$N=1$:} $\mathrm{PR}\approx 3$ (bounded).\quad
              \textbf{$N=2$:} $\mathrm{PR}$ up to $\sim 20$.
        \item The \textbf{$P(v)$ heatmap} plots, for every bright polaritonic
              state (x, energy-ordered), its $P(v)$ over sectors $v$ (y),
              colored by value \refmark{2}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Introducing disorder}
%=============================================================================
\begin{frame}{Why Disorder?}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \small
      Real molecular ensembles are \textbf{inhomogeneous}: each molecule has a
      slightly different transition energy (static \emph{energy disorder}).
      This work frames what disorder does to a cavity-coupled ensemble of
      two-level molecules:
      \begin{block}{Disordered Holstein--Tavis--Cummings (with vibrations)\refmark{4}}
      \scriptsize
      Each molecule is a two-level system on a displaced harmonic potential
      (Holstein coupling). Central result: \textbf{disorder enhances}
      excitation transfer and vibrational entanglement, and dark states
      \emph{acquire photon weight} (brighten). Effect peaks when disorder
      $W\sim g_c$.
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[height=0.32\textheight]{fig_disorder_twolevel.png}\\[0.15em]
      \figcap{Two-level molecule: displaced harmonic surfaces, spacing $\nu$.}\\[0.35em]
      \begin{tikzpicture}[scale=0.52, every node/.style={font=\scriptsize}]
        \draw[dashed,gray] (-0.3,2) -- (6.3,2);
        \node[capgray,anchor=west] at (6.3,2) {$\omega$};
        \foreach \x/\o in {0/0.5, 1.4/-0.3, 2.8/0.15, 4.2/-0.55, 5.6/0.35} {
          \draw[thick] (\x,0) -- (\x+0.8,0);
          \draw[thick,purple!70] (\x,2+\o) -- (\x+0.8,2+\o);
          \draw[->,purple!45,thin] (\x+0.4,0.15) -- (\x+0.4,{1.85+\o});
        }
        \node[below] at (3,-0.4) {$N$ molecules, random $\epsilon_n$};
      \end{tikzpicture}\\[0.2em]
      \figcap{Many two-level molecules with disordered transition energies $\omega+\epsilon_n$.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Disorder: The Equations}
  \small
  \textbf{Disordered Holstein--Tavis--Cummings model}\refmark{4}:
  \[
    \hat H=\hat H_{\mathrm{TC}}+\hat H_{\mathrm{vib}}+\hat H_{\mathrm{H}}+\hat H_{\mathrm{dis}},
    \qquad
    \boxed{\ \hat H_{\mathrm{dis}}=\sum_n \epsilon_n\!\!\sum_{r=+,-}\hat\sigma_{n,r}^\dagger\hat\sigma_{n,r}\ }
  \]
  with energy offsets $\epsilon_n$ of mean $0$ and variance $W^2$ (Gaussian). For
  each sublevel $r=+,-$,
  \[
    \hat\sigma_{n,r}^\dagger = |E_r\rangle\langle A|_n\ (A\!\to\!E_r), \qquad
    \hat\sigma_{n,r} = |A\rangle\langle E_r|_n\ (E_r\!\to\!A)
  \]
  are the \textbf{raising/lowering operators between $|A\rangle$ and $|E_r\rangle$
  on the $n$-th molecule}, so $\hat\sigma_{n,r}^\dagger\hat\sigma_{n,r}=|E_r\rangle\langle E_r|_n$
  is the projector onto sublevel $r$, giving
  \[
    \hat H_{\mathrm{dis}}=\sum_n \epsilon_n\big(|E_+\rangle\langle E_+|_n+|E_-\rangle\langle E_-|_n\big),
  \]
  which shifts each molecule's excited-sublevel energies by its own random
  offset $\epsilon_n$. Consequences quantified in the paper:
  \begin{itemize}\small
    \item Dark states gain photon weight
          $\sum_{\mathrm{dark}}|\langle d|1_{\mathrm{ph}}\rangle|^2\approx W^2/g_c^2$
          $\Rightarrow$ they \textbf{appear in absorption}.
    \item Polaritons stay protected while $g_c$ beats disorder: crossover at
          $W\sim 2g_c$ (with $g_c=\Omega/2$).
  \end{itemize}
\end{frame}

\section{Disorder in our model \& results}
%=============================================================================
\begin{frame}{How Disorder Enters Our Model}
  \small
  We add \textbf{Gaussian energy disorder} to the two JT molecules'
  transition energies:
  \[
    \epsilon_i \sim \mathcal N(\epsilon,\ W^2),\qquad i=1,2,\qquad \epsilon=7.0~\text{eV}.
  \]
  Per disorder \emph{realization} we build $\hat H$ in the
  $(n_{\mathrm{ex}}=1,\ j=-1)$ sector, diagonalize, take the intensity of each
  eigenstate as $|\langle i_0|\psi\rangle|^2$, broaden with a Lorentzian (FWHM, $\gamma=0.025$~eV), and
  \textbf{average over realizations}.
  \begin{block}{Energy scales}

\begin{frame}{Summary}
  \small
  \begin{enumerate}
    \item \textbf{Reproduced} the single-molecule cavity-JT polariton spectrum
          (LP/UP, bounded $v$) and the two-molecule collective spectrum +
          $P(v)$ heatmap (unbounded cascade, $\mathrm{PR}\sim20$).
    \item \textbf{Introduced disorder} following the arrowhead-matrix and
          disordered Holstein--Tavis--Cummings frameworks: Gaussian excitation-energy
          disorder $\epsilon_i\sim\mathcal N(\epsilon,\sigma^2)$ used.
    \item \textbf{Implemented \& analyzed} disorder in our $N=2$ model:
          realizations, representative cases (A--H), and $P(v)$ heatmaps with an
          LP/UP energy overlay.
    \item \textbf{Key result:} destroys the observed cascade depends on whether it pushes the bright polariton closer to the dense manifold of otherwise-dark states, or pulls it away — governed by proximity to cavity resonance and to the collective coupling scale, not disorder magnitude alone.

  \end{enumerate}
\end{frame}
